<programa> ::= <principal> 
            | <globales> <principal> 
            | <funciones> <principal>
            | <globales> <funciones> <principal>

<principal> ::= <tipo_principal> 'principal' <bloque>

<globales> ::= <declaracion_global> | <declaracion_global> <globales>
<declaracion_global> ::=  <creacion_variable>

<funciones> ::= <funcion> | <funcion> <funciones>
<funcion> ::= <tipo_retorno> <id> <parametros> <bloque> 
<llamar_funcion> ::= <id> <argumentos>

<parametros> ::= <parentesisR_izquierdo> <parentesisR_derecho> | <parentesisR_izquierdo> <aux_parametros> <parentesisR_derecho>
<aux_parametros> ::= <parametro> | <parametro> <coma> <aux_parametros>
<parametro> ::= <tipo> <id>

<argumentos> ::= <parentesisR_izquierdo> <parentesisR_derecho> | <parentesisR_izquierdo> <auxArgumentos> <parentesisR_derecho>
<auxArgumentos> ::= <expresion> | <expresion> <coma> <auxArgumentos>

<expresion>::= <expresion_aritmetica> | <comparacion> | <operacion_logica> | <boolD> | <stringD> | <charD> | <arreglo> | <llamar_funcion>

<expresion_aritmetica> ::= <expresion_aritmetica> <modulo> <expresion_mul_div> | <expresion_mul_div>

<expresion_mul_div> ::= <expresion_mul_div> <multiplicacion> <expresion_suma_resta> 
                        | <expresion_mul_div> <division_decimal> <expresion_suma_resta> 
                        | <expresion_mul_div> <division_entera> <expresion_suma_resta> 
                        | <expresion_suma_resta>

<expresion_suma_resta> ::= <expresion_suma_resta> <signo> <expresion_potencia> | <expresion_potencia>

<expresion_potencia> ::= <expresion_potencia> <potencia> <base> | <base>

<base> ::= <id> | <id> '++' | <id> '--' | <numero_entero> | <numero_flotante> | <llamar_funcion> | <acceso_arreglo> | <parentesisR_izquierdo> <expresion_aritmetica> <parentesisR_derecho>

<condicion> ::= <expresion>

<comparacion> ::= <parentesisC_izquierdo> <expresion> <parentesisC_derecho> <operador_relacional> <parentesisC_izquierdo> <expresion> <parentesisC_derecho>

<operacion_logica> ::= <parentesisR_izquierdo> <condicion> <parentesisR_derecho> <operadorL_binario> <parentesisR_izquierdo> <condicion> <parentesisR_derecho> 
                        | <negacion> <parentesisR_izquierdo> <condicion> <parentesisR_derecho>

<bloque> ::= <abrir_bloque> <lista_cosas> <cerrar_bloque>

<lista_cosas> ::= <cosa> | <cosa> <lista_cosas>
<cosa> ::= <creacion_variable> 
            | <asignacion_variable> | <asignacion_arreglo> | <sentencia_if> | <sentencia_while> | <sentencia_for> 
            | <llamar_funcion> <fin_expresion> | <incrementoV> | <sentencia_break> | <sentencia_return>
            | <decrementoV> | <funcion_read> | <funcion_write> | <comentario>

<creacion_variable> ::= <variable> <parametro> <fin_expresion> 
                        | <variable> <parametro> <asignacionV> <expresion> <fin_expresion> 
                        | <variable> <tipo_arr> <id> <parentesisC_izquierdo> <entero_positivo> <parentesisC_derecho> <fin_expresion> 
                        | <variable> <tipo_arr> <id> <asignacionV> <arreglo> <fin_expresion> 

<asignacion_variable> ::= <id> <asignacionV> <expresion> <fin_expresion>
<asignacion_arreglo> ::= <acceso_arreglo> <asignacionV> <expresion> <fin_expresion>

<incrementoV> ::= <id> '++' <fin_expresion>
<decrementoV> ::= <id> '--' <fin_expresion>

<if> ::= 'if'
<elif> ::= 'elif'
<else> ::= 'else'
<while> ::= 'while'
<for> ::= 'for'
<break> ::= 'break'
<return> ::= 'return'

<sentencia_if> ::= <if> <parentesisR_izquierdo> <condicion> <parentesisR_derecho> <bloque> <lista_elif> <sentencia_else>

<lista_elif> ::= <sentencia_elif> <lista_elif> | ""
<sentencia_elif> ::= <elif> <parentesisR_izquierdo> <condicion> <parentesisR_derecho> <bloque>

<sentencia_else> ::= <else> <bloque> | ""

<sentencia_while> ::= <while> <parentesisR_izquierdo> <condicion> <parentesisR_derecho> <bloque>

<sentencia_for> ::= <for> <parentesisR_izquierdo> <asignacion_for_init> <fin_expresion> <condicion> <fin_expresion> <expresion_for_final> <parentesisR_derecho> <bloque>
<asignacion_for_init> ::= <id> <asignacionV> <expresion>
<expresion_for_final> ::= <id> <asignacionV> <expresion> | <id> '++' | <id> '--'

<sentencia_return> ::= <return> <expresion> <fin_expresion> | <return> <fin_expresion>

<sentencia_break> ::= <break> <fin_expresion>

<funcion_read> ::= <read> <parentesisR_izquierdo> <id> <parentesisR_derecho> <fin_expresion>

<funcion_write> ::= <write> <parentesisR_izquierdo> <imprimir> <parentesisR_derecho> <fin_expresion>
<imprimir> ::= <id> | <stringD> | <numero_entero> | <numero_flotante> | <boolD>

<write> ::= 'write'
<read> ::= 'read'

<arreglo> ::= <abrir_bloque> <elemento_lista> <cerrar_bloque> | <abrir_bloque> <cerrar_bloque>
<acceso_arreglo> ::= <id> <parentesisC_izquierdo> <expresion> <parentesisC_derecho>  
<elemento_lista> ::= <elemento_arr> | <elemento_arr> <coma> <elemento_lista> 
<elemento_arr> ::= <numero_entero> | <numero_flotante>

<tipo> ::= <tipo_retorno> | 'string'
<tipo_principal> ::= 'void'
<tipo_arr> ::= 'int' | 'float'
<tipo_retorno> ::= 'char' | 'int' | 'float' | 'bool'

<comentario> ::= <comentario_s> | <comentario_multilinea>

<comentario_s> ::= '|' <texto_linea>
<texto_linea> ::= <caracterL> | <caracterL> <texto_linea>
<caracterL> ::= [^\n]

<comentario_multilinea> ::= <comentarioM_izquierdo> <texto_multi> <comentarioM_derecha>
<texto_multi> ::= <caracterM> | <caracterM> <texto_multi>
<caracterM> ::= [^!]

<comentarioM_izquierdo> ::= '¡'
<comentarioM_derecha> ::= '!'

<id> ::= [_A-Za-z][_A-Za-z0-9]*
<boolD> ::= 'true' | 'false'
<stringD> ::= "[^"]*"
<charD> ::= '[^|!¡']'

<signo> ::= '-'|'+'
<cero> ::= '0'
<digito> ::= [0-9]
<digitoN_Cero> ::= [1-9]
<numero_entero> ::= <cero> | <signo>? <digitoN_Cero> <digito>*
<entero_positivo> ::= <cero> | <digitoN_Cero> <digito>*

<ceroF> ::= '0.0'
<numero_flotante> ::= <ceroF> 
                    | <signo>? <cero> <punto> <digito>* <digitoN_Cero> 
                    | <signo>? <digitoN_Cero> <digito>* <punto> <cero>
                    | <signo>? <digitoN_Cero> <digito>* <punto> <digito>* <digitoN_Cero> 
                    
<parentesisR_izquierdo> ::= 'є:' 
<parentesisR_derecho> ::= ':э' 
<parentesisC_izquierdo> ::= 'ʃ:'
<parentesisC_derecho> ::= ':ʅ'
<abrir_bloque> ::= '¿:'
<cerrar_bloque> ::= ':?'
<punto> ::= '.'
<coma> ::= ','

<suma> ::= '+'
<resta> ::= '-'
<multiplicacion> ::= '*'
<division_decimal> ::= '/'
<division_entera> ::= '//'
<modulo> ::= 'mod' 
<potencia> ::= 'pot'

<operador_relacional> ::= '>' | '<' | '>=' | '<=' | '==' | '!='
<operadorL_binario> ::= 'λ' | 'θ'
<negacion> ::= 'Σ'

<fin_expresion> ::= '»'

<variable> ::= 'val'
<asignacionV> ::= 'Ͱ'
